cmake_minimum_required(VERSION 3.16)
project(equihashverify LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
find_package(Threads REQUIRED)
file(GLOB crypto_sources crypto/*.cpp)
add_library(equihashverify STATIC ${crypto_sources} util.cpp utilstrencodings.cpp uint256.cpp arith_uint256.cpp random.cpp support/cleanse.cpp)
target_include_directories(equihashverify PUBLIC .)
target_compile_options(equihashverify PRIVATE -UNDEBUG -include stdexcept)
target_compile_definitions(equihashverify PRIVATE HAVE_DECL_STRNLEN=1 HAVE_BYTESWAP_H=1)
target_link_libraries(equihashverify PUBLIC sodium boost_system Threads::Threads)

include(CTest)
if(BUILD_TESTING)
  add_executable(verify_tests tests/verify.cpp)
  target_link_libraries(verify_tests PRIVATE equihashverify)
  target_compile_options(verify_tests PRIVATE -UNDEBUG -include stdexcept)
  add_test(NAME native_proofs COMMAND verify_tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/proofs.tsv)
  add_executable(avx2_hash_tests tests/avx2_hashes.cpp)
  target_link_libraries(avx2_hash_tests PRIVATE equihashverify)
  target_compile_options(avx2_hash_tests PRIVATE -UNDEBUG -include stdexcept)
  add_test(NAME avx2_leaf_hashes COMMAND avx2_hash_tests)
  set_tests_properties(avx2_leaf_hashes PROPERTIES SKIP_RETURN_CODE 77)
endif()
